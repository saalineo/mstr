cmake_minimum_required(VERSION 3.16)
project(mstr C CXX)

option(MSTR_CHECKED "Enable checked mode" OFF)
option(MSTR_SANITIZE "ASan+UBSan" OFF)
option(MSTR_TSAN "ThreadSanitizer (not combinable with MSTR_SANITIZE)" OFF)

set(CMAKE_C_STANDARD 99)
set(CMAKE_C_STANDARD_REQUIRED ON)
set(CMAKE_C_EXTENSIONS OFF)
if(NOT CMAKE_CXX_STANDARD)
  set(CMAKE_CXX_STANDARD 17)
endif()
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)

# Directory-wide options must come BEFORE the targets they should apply to.
if(MSVC)
  add_compile_options(/W4)
else()
  add_compile_options(-Wall -Wextra -Wpedantic -Wconversion -Werror)
endif()
if(MSTR_SANITIZE AND MSTR_TSAN)
  message(FATAL_ERROR "MSTR_SANITIZE and MSTR_TSAN are mutually exclusive")
endif()
if(MSTR_SANITIZE AND NOT MSVC)
  add_compile_options(-fsanitize=address,undefined -fno-sanitize-recover=all -fno-omit-frame-pointer)
  add_link_options(-fsanitize=address,undefined)
endif()
if(MSTR_TSAN AND NOT MSVC)
  add_compile_options(-fsanitize=thread -fno-omit-frame-pointer)
  add_link_options(-fsanitize=thread)
endif()

add_library(mstr STATIC src/mstr.c)
target_include_directories(mstr PUBLIC include)
if(MSTR_CHECKED)
  target_compile_definitions(mstr PUBLIC MSTR_CHECKED)
endif()

enable_testing()
add_executable(test_m1 tests/test_m1.c)
target_link_libraries(test_m1 mstr)
add_test(NAME m1 COMMAND test_m1)
add_executable(test_header_cpp tests/test_header_cpp.cpp)
target_link_libraries(test_header_cpp mstr)
add_test(NAME header_cpp COMMAND test_header_cpp)

add_executable(test_m2 tests/test_m2.c)
target_link_libraries(test_m2 mstr)
add_test(NAME m2 COMMAND test_m2)
add_executable(test_m2_strict tests/test_m2.c)
target_compile_definitions(test_m2_strict PRIVATE MSTR_STRICT_SCOPE)
target_link_libraries(test_m2_strict mstr)
add_test(NAME m2_strict_scope COMMAND test_m2_strict)
if(NOT MSVC)
  find_package(Threads REQUIRED)
  add_executable(test_thread_exit tests/test_thread_exit.c)
  target_link_libraries(test_thread_exit mstr Threads::Threads)
  add_test(NAME thread_exit COMMAND test_thread_exit)
endif()

# Checked-mode variant: validates scope serials on every access.
add_library(mstr_checked STATIC src/mstr.c)
target_include_directories(mstr_checked PUBLIC include)
target_compile_definitions(mstr_checked PUBLIC MSTR_CHECKED)
if(NOT MSVC)
  add_executable(test_death tests/test_death.c)
  target_link_libraries(test_death mstr_checked Threads::Threads)
  add_test(NAME death_checked COMMAND test_death)
  add_executable(test_m2_checked tests/test_m2.c)
  target_link_libraries(test_m2_checked mstr_checked)
  add_test(NAME m2_checked COMMAND test_m2_checked)
  set_tests_properties(m2_checked PROPERTIES ENVIRONMENT MSTR_NO_ROOT_WARNING=1)

endif()

# libFuzzer harnesses (clang only): cmake -DMSTR_FUZZ=ON -DCMAKE_C_COMPILER=clang
option(MSTR_FUZZ "Build libFuzzer harnesses" OFF)
if(MSTR_FUZZ)
  foreach(h slice trim replace split utf8 pipe)
    add_executable(fuzz_${h} fuzz/fuzz_${h}.c)
    target_include_directories(fuzz_${h} PRIVATE fuzz)
    target_link_libraries(fuzz_${h} mstr_checked)
    target_compile_options(fuzz_${h} PRIVATE -fsanitize=fuzzer-no-link,address,undefined -g)
    target_link_options(fuzz_${h} PRIVATE -fsanitize=fuzzer,address,undefined)
    add_test(NAME fuzz_smoke_${h} COMMAND fuzz_${h} -max_total_time=10 -max_len=512)
  endforeach()
endif()

# Examples (also the README programs) and their tests
if(NOT MSVC)
  foreach(e readme_malloc readme_mstr cli_trim csv_split log_filter)
    add_executable(ex_${e} examples/${e}.c)
    target_link_libraries(ex_${e} mstr)
  endforeach()
  add_test(NAME ex_readme_same_output
           COMMAND sh -c "for i in '' '  hello   world  ' 'x' '   '; do a=$($<TARGET_FILE:ex_readme_malloc> \"$i\"); b=$($<TARGET_FILE:ex_readme_mstr> \"$i\"); [ \"$a\" = \"$b\" ] || exit 1; done")
  add_test(NAME ex_cli_trim
           COMMAND sh -c "printf '  a  \\n\\n  b c \\n' | $<TARGET_FILE:ex_cli_trim> 2>/dev/null | tr '\\n' '|' | grep -qx 'a|b c|'")
  add_test(NAME ex_csv_split
           COMMAND sh -c "printf 'a, b, 1.5\\nc,d , 2\\n' | $<TARGET_FILE:ex_csv_split> | grep -qx 'rows=2 sum=3.50'")
  add_test(NAME ex_log_filter
           COMMAND sh -c "printf '2024-05-01 12:00:01 ERROR db: timeout\\n2024-05-01 12:00:02 INFO ok\\n' | $<TARGET_FILE:ex_log_filter> error | grep -qx '\\[12:00:01\\] db: timeout'")
  foreach(t ex_readme_same_output ex_cli_trim ex_csv_split ex_log_filter)
    set_tests_properties(${t} PROPERTIES ENVIRONMENT MSTR_NO_ROOT_WARNING=1)
  endforeach()
  find_package(Python3 COMPONENTS Interpreter)
  if(Python3_FOUND)
    add_test(NAME readme_in_sync COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/check_readme.py)
    add_test(NAME docs_programs_build_and_run
             COMMAND ${Python3_EXECUTABLE} ${CMAKE_SOURCE_DIR}/tests/check_guide.py
                     ${CMAKE_C_COMPILER} ${CMAKE_CXX_COMPILER} ${CMAKE_SOURCE_DIR})
  endif()
endif()

# Benchmarks vs SDS and hand-written malloc code: cmake -DMSTR_BENCH=ON, then run ./bench [--md]
option(MSTR_BENCH "Build benchmarks (vendored SDS in bench/third_party/sds)" OFF)
if(MSTR_BENCH AND NOT MSVC)
  add_library(mstr_bench STATIC src/mstr.c)
  target_include_directories(mstr_bench PUBLIC include)
  target_compile_definitions(mstr_bench PRIVATE MSTR_STATS NDEBUG)
  target_compile_options(mstr_bench PRIVATE -O2)
  add_library(sds_bench STATIC bench/third_party/sds/sds.c)
  target_include_directories(sds_bench PRIVATE bench/sds_shim)
  target_include_directories(sds_bench SYSTEM PUBLIC bench/third_party/sds) # third-party headers: no -Werror noise
  target_compile_options(sds_bench PRIVATE -O2 -w) # vendored, unmodified upstream code
  add_executable(bench bench/bench.c)
  target_compile_options(bench PRIVATE -O2)
  target_link_libraries(bench mstr_bench sds_bench)
endif()
